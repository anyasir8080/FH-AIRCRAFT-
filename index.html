CREATE OR REPLACE FUNCTION public.add_fleet_utilization( p_aircraft text, p_fh numeric, p_fc integer ) RETURNS boolean LANGUAGE plpgsql SECURITY INVOKER SET search_path = '' AS $$ BEGIN IF p_aircraft IS NULL OR p_fh IS NULL OR p_fc IS NULL OR p_fh < 0 OR p_fc < 0 OR (p_fh = 0 AND p_fc = 0) OR p_fh <> round(p_fh, 2) THEN RAISE EXCEPTION 'Invalid utilization values'; END IF; UPDATE public.fleet SET fh = fh + p_fh, fc = fc + p_fc, updated_at = now() WHERE aircraft = p_aircraft; RETURN FOUND; END; $$; REVOKE ALL ON FUNCTION public.add_fleet_utilization( text,numeric,integer ) FROM PUBLIC; GRANT EXECUTE ON FUNCTION public.add_fleet_utilization( text,numeric,integer ) TO anon, authenticated;
